'use strict';

// Test tooling, not production code: recomputes the hashes a capture recorded and reports every difference, so that a
// tampered, missing or stale retained file is a named failure (C044 F9, L-0133).
const fs = require('node:fs');
const path = require('node:path');
const zlib = require('node:zlib');
const { execFileSync } = require('node:child_process');
const { createHash } = require('node:crypto');

const sha = (buffer) => createHash('sha256').update(buffer).digest('hex');
const crlfToLf = (buffer) => Buffer.from(buffer.toString('latin1').replace(/\r\n/g, '\n'), 'latin1');

// EVERY manifest of EVERY capture (red, green, coverage, mutants, browser and the fix captures): a capture manifest is
// recognised by its shape, so no folder (the raw folder holds the green, coverage, mutants and browser manifests) is skipped.
function manifestsUnder(directory) {
  const found = [];
  const walk = (dir) => {
    for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
      const full = path.join(dir, entry.name);
      if (entry.isDirectory()) walk(full);
      else if (entry.name.endsWith('.json')) {
        const body = JSON.parse(fs.readFileSync(full, 'utf8'));
        if (body.schemaVersion === 1 && body.transcripts && Array.isArray(body.inputs)) found.push(full);
      }
    }
  };
  if (fs.existsSync(directory)) walk(directory);
  return found.sort();
}

// The bytes of a pinned input as they are today (default) or as they were in a given commit (a sealed capture).
const workingReader = (repository) => (relative) => fs.readFileSync(path.join(repository, relative));
const gitReader = (commit, repository) => (relative) => execFileSync('git', ['--no-pager', 'cat-file', 'blob', `${commit}:${relative.replace(/\\/g, '/')}`],
  { cwd: repository, maxBuffer: 16 * 1024 * 1024, windowsHide: true, stdio: ['ignore', 'pipe', 'ignore'] });
const U060_COMPLETION_AUTHORITY_SHA256 = 'b91feb82e2b4cd8659468407551b1c2d2c32c4c66f65f1c6468704f2865348ac';
const manifestReader = (manifestFile, repository) => {
  const checker = path.join(__dirname, 'u060-completion-authority.cjs');
  if (sha(fs.readFileSync(checker)) !== U060_COMPLETION_AUTHORITY_SHA256) throw new Error('CHECKER_BYTES_CHANGED capability\\C043\\tests\\u060-completion-authority.cjs');
  return require('./u060-completion-authority.cjs').manifestReader(manifestFile, repository);
};

// Every input must have the recorded bytes (size, SHA-256 and LF-normalized SHA-256) and every transcript must hash and
// unzip to what was recorded.
function verifyManifest(manifestPath, repository, read = workingReader(repository)) {
  const problems = [];
  const manifest = JSON.parse(fs.readFileSync(manifestPath, 'utf8'));
  const here = path.dirname(manifestPath);
  for (const input of manifest.inputs) {
    let bytes;
    try { bytes = read(input.path); } catch { problems.push(`${manifest.name}: input ${input.path} is missing`); continue; }
    const exact = sha(bytes) === input.rawSha256 && bytes.length === input.bytes && sha(crlfToLf(bytes)) === input.crlfToLfSha256;
    const external = !input.path.replace(/\\/g, '/').startsWith('capability/C043/');
    if (!exact && !(external && sha(crlfToLf(bytes)) === input.crlfToLfSha256)) problems.push(`${manifest.name}: input ${input.path} differs from the recorded bytes`);
  }
  for (const [stream, info] of Object.entries(manifest.transcripts)) {
    const file = path.join(here, info.path);
    if (!fs.existsSync(file)) { problems.push(`${manifest.name}: ${stream} transcript is missing`); continue; }
    const gz = fs.readFileSync(file);
    if (sha(gz) !== info.gzipSha256) { problems.push(`${manifest.name}: ${stream} transcript bytes changed`); continue; }
    let plain;
    try { plain = zlib.gunzipSync(gz); } catch { problems.push(`${manifest.name}: ${stream} transcript does not unzip`); continue; }
    if (sha(plain) !== info.sha256 || plain.length !== info.bytes) problems.push(`${manifest.name}: ${stream} content differs from the recorded hash`);
  }
  return problems;
}

module.exports = Object.freeze({ gitReader, manifestsUnder, verifyManifest, workingReader, manifestReader, sha });
