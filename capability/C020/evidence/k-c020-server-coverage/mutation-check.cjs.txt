// A mutant names its target by text; without an explicit occurrence the target must be unique, so no mutant is ambiguous.
function apply(source, [, find, replacement, nth]) {
  const parts = source.split(find);
  const occurrences = parts.length - 1;
  if (nth === undefined ? occurrences !== 1 : nth >= occurrences) throw new Error(`The mutation target matches ${occurrences} time(s): ${find}`);
  const at = nth ?? 0;
  return parts.slice(0, at + 1).join(find) + replacement + parts.slice(at + 1).join(find);
}

function runSuite(root) {
  const env = { ...process.env, K_C020_NAMESPACE_ROOT: root };
  delete env.NODE_TEST_CONTEXT;
  delete env.NODE_V8_COVERAGE;
  const run = spawnSync(process.execPath, ['--test', '--test-concurrency=1', '--test-timeout=60000', '--test-reporter=tap', ...tests],
    { cwd: repo, env, encoding: 'utf8', timeout: 300000, maxBuffer: 32 * 1024 * 1024 });
  const count = (name) => Number((new RegExp(`^# ${name} (\\d+)`, 'm').exec(run.stdout) || [])[1]);
  return { exit: run.status, signal: run.signal, tests: count('tests'), pass: count('pass'), fail: count('fail'),
    failing: [...run.stdout.matchAll(/^not ok \d+ - (.*)$/gm)].map((match) => match[1]).slice(0, 3) };
}

fs.mkdirSync(path.join(namespace, 'tests'), { recursive: true });

for (const name of ['server.cjs', 'profiles.cjs', 'contract.js', 'inspector.js', 'page.js', 'index.html', 'styles.css', 'README.md', 'evidence.json']) {
  fs.copyFileSync(path.join(repo, 'capability', 'C020', name), path.join(namespace, name));
}
for (const name of ['contract-guard.js', 'approved-contract.js']) {
  fs.copyFileSync(path.join(repo, 'capability', 'C001', name), path.join(scratch, 'capability', 'C001', name));
}
const original = fs.readFileSync(path.join(namespace, 'server.cjs'), 'utf8');
for (const mutant of mutants) apply(original, mutant);
const baseline = runSuite(namespace);
const results = [];
if (baseline.exit === 0) {
  for (const mutant of mutants) {
    fs.writeFileSync(path.join(namespace, 'server.cjs'), apply(original, mutant));
    const outcome = runSuite(namespace);
    results.push({ id: mutant[0], killed: outcome.exit !== 0, ...outcome });
    const line = `${outcome.exit !== 0 ? 'KILLED  ' : 'SURVIVED'} ${mutant[0]} (${outcome.failing[0] || `exit ${outcome.exit}`})`;
    console.log(line);
    fs.appendFileSync(`${scratch}.progress.log`, `${line}\n`);
  }
}
fs.rmSync(scratch, { recursive: true, force: true });
const summary = { total: results.length, killed: results.filter((item) => item.killed).length, survived: results.filter((item) => !item.killed).length };
const record = { schemaVersion: 1, tool: 'k-c020-mutation-check', node: process.version, serverSha256: sha(original),
  testFiles: tests.map((file) => path.basename(file)), baseline, summary, mutants: results };
fs.writeFileSync(resultFile, `${JSON.stringify(record, null, 2)}\n`, { flag: 'wx' });
console.log(JSON.stringify({ baseline: baseline.exit, ...summary }));
process.exitCode = baseline.exit === 0 && summary.survived === 0 ? 0 : 1;
