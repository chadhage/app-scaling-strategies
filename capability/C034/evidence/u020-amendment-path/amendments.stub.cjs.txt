function applySurfaceEdit(bytes) {
  const text = bytes.toString('utf8');
  const anchor = `${SURFACE_EDIT.before}\r\n`;
  if (text.split(anchor).length !== 2) throw Object.assign(new Error('SURFACE_EDIT_ANCHOR'), { code: 'SURFACE_EDIT_ANCHOR' });
  return Buffer.from(text.replace(anchor, `${SURFACE_EDIT.after}\r\n`));
}
