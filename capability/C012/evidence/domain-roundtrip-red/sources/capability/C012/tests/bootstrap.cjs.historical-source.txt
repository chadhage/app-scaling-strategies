function copy(relative, commit, tag, role) {

  const blob = git(['show', commit + ':' + gitPath]);
  const lf = Buffer.from(raw.toString('utf8').replace(/\r\n/g, '\n'));

  const target = path.join(root, relative);
  fs.mkdirSync(path.dirname(target), { recursive: true });

    if (!fs.readFileSync(target).equals(raw)) throw new Error('Private existing copy differs: ' + relative);

}
