'use strict';
const fs=require('node:fs'),path=require('node:path'),assert=require('node:assert/strict');
const {run,remaining}=require('./budget.cjs');
const {physical,copyTree,list}=require('./source.cjs');
const {representation}=require('./representations.cjs');
const {sha}=require('./integrity.cjs');
const ns=path.resolve(__dirname,'..'),freezeFile=process.env.C012_CURRENT_FREEZE;
const output=process.env.C012_CURRENT_FRESH_OUTPUT;
assert.ok(freezeFile&&output&&path.isAbsolute(freezeFile)&&path.isAbsolute(output));
remaining(60000);fs.mkdirSync(output,{recursive:false});
const freeze=JSON.parse(fs.readFileSync(freezeFile)),work=path.join(process.env.C012_PRIVATE_ROOT,'w-current-v2');
for(const row of freeze.sources)assert.equal(sha(fs.readFileSync(path.join(ns,row.path))),row.sha256);
fs.mkdirSync(work);const identity=physical(work),reports=[];
try{
  for(const kind of ['lf','crlf']){
    remaining(60000);
    const root=path.join(work,kind),target=path.join(root,'capability','C012');
    fs.mkdirSync(path.dirname(target),{recursive:true});copyTree(ns,target);
    for(const row of freeze.sources){
      const bytes=fs.readFileSync(path.join(path.dirname(freezeFile),'sources',row.path));
      fs.writeFileSync(path.join(target,row.path),representation(row.path,bytes,kind));
    }
    for(const row of require('../imports.json').entries){
      const file=path.join(root,row.path);fs.mkdirSync(path.dirname(file),{recursive:true});
      const bytes=fs.readFileSync(path.join(path.dirname(freezeFile),'imports',row.path));
      fs.writeFileSync(file,representation(row.path,bytes,kind),{flag:'wx'});
    }
    const before=list(target),result=run(['capability\\C012\\tests\\current-v2-suite.cjs'],{cwd:root,env:{C012_FRESH_KIND:kind},timeout:1500000});
    for(const stream of ['stdout','stderr'])fs.writeFileSync(path.join(output,kind+'.'+stream+'.raw'),result[stream]||'',{flag:'wx'});
    assert.deepEqual(list(target),before);
    const text=(result.stdout||'').toString(),summary=Object.fromEntries(['tests','pass','fail','cancelled','skipped','todo'].map(key=>[key,Number(new RegExp('^# '+key+' (\\d+)$','m').exec(text)?.[1]??-1)]));
    reports.push({kind,sources:before,exit:result.status,error:result.error?.code||null,signal:result.signal,summary,
      stdoutSha256:sha(result.stdout||''),stderrSha256:sha(result.stderr||''),historicalReuseClaim:'fresh representation comparison only, not a second native/resource campaign'});
    assert.equal(result.error,undefined);assert.equal(result.signal,null);assert.equal(result.status,0,String(result.stderr));assert.equal(summary.tests,summary.pass);assert.ok(summary.tests>43);
  }
}finally{
  assert.deepEqual(physical(work),identity);fs.rmSync(work,{recursive:true,force:false});
  fs.writeFileSync(path.join(output,'report.json'),JSON.stringify({schema:'C012-current-v2-complete-fresh-LF-CRLF/2',reports,
    freezeSha256:sha(fs.readFileSync(freezeFile)),workspaceRemoved:!fs.existsSync(work),pass:reports.length===2&&reports.every(row=>row.exit===0&&!row.error&&!row.signal)},null,2)+'\n',{flag:'wx'});
}
console.log(JSON.stringify({completeFreshKinds:reports.map(row=>row.kind),summaries:reports.map(row=>row.summary),removed:true}));
