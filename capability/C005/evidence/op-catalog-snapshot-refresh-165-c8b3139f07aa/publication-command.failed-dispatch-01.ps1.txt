function Run-Tool([string]$Executable, [string[]]$Arguments) {
    $start = [Diagnostics.ProcessStartInfo]::new()
    $start.FileName = $Executable
    $start.WorkingDirectory = (Get-Location).Path
    $start.UseShellExecute = $false
    $start.RedirectStandardOutput = $true
    $start.RedirectStandardError = $true
    foreach ($argument in $Arguments) { $start.ArgumentList.Add($argument) }
    $process = [Diagnostics.Process]::new()
    $process.StartInfo = $start
    $out = [IO.MemoryStream]::new()
    $err = [IO.MemoryStream]::new()
    try {
        if (-not $process.Start()) { throw "Cannot start $Executable." }
        $outTask = $process.StandardOutput.BaseStream.CopyToAsync($out)
        $errTask = $process.StandardError.BaseStream.CopyToAsync($err)
        $process.WaitForExit()
        $outTask.GetAwaiter().GetResult()
        $errTask.GetAwaiter().GetResult()
        $outBytes = $out.ToArray()
        $errBytes = $err.ToArray()
        $record = [ordered]@{
            executable = $Executable; arguments = $Arguments; exitCode = $process.ExitCode
            stdoutOffset = $rawOut.Length; stdoutBytes = $outBytes.Length
            stdoutSha256 = [Convert]::ToHexString([Security.Cryptography.SHA256]::HashData($outBytes)).ToLowerInvariant()
            stderrOffset = $rawErr.Length; stderrBytes = $errBytes.Length
            stderrSha256 = [Convert]::ToHexString([Security.Cryptography.SHA256]::HashData($errBytes)).ToLowerInvariant()
        }
        $commands.Add($record)
        $rawOut.Write($outBytes, 0, $outBytes.Length)
        $rawErr.Write($errBytes, 0, $errBytes.Length)
        $outText = [Text.Encoding]::UTF8.GetString($outBytes)
        $errText = [Text.Encoding]::UTF8.GetString($errBytes)
        if ($process.ExitCode -ne 0) { throw "$Executable $($Arguments -join ' ') failed ($($process.ExitCode)): $outText$errText" }
        return $outText
    } finally {
        $out.Dispose(); $err.Dispose(); $process.Dispose()
    }
}

function File-Hash([string]$Path) {
    return (Get-FileHash -LiteralPath $Path -Algorithm SHA256).Hash.ToLowerInvariant()
}
function Assert-Index($Before) {
    $after = Check @('--index')
    foreach ($key in @('entries', 'flags', 'stageBytes', 'flagBytes',
        'unownedLsFilesStageRawByteSha256', 'unownedLsFilesFlagsRawByteSha256')) {
        if ($Before.$key -ne $after.$key) { throw "Unowned index bytes or flags changed: $key" }
    }
}
function Save-Raw([string]$Path, [byte[]]$Bytes) {
    $stream = [IO.File]::Open($Path, [IO.FileMode]::CreateNew)
    try {
        $gzip = [IO.Compression.GZipStream]::new($stream, [IO.Compression.CompressionLevel]::Optimal, $true)
        try { $gzip.Write($Bytes, 0, $Bytes.Length) } finally { $gzip.Dispose() }
    } finally { $stream.Dispose() }
    return [ordered]@{ path = $Path; bytes = $Bytes.Length
        sha256 = [Convert]::ToHexString([Security.Cryptography.SHA256]::HashData($Bytes)).ToLowerInvariant()
        gzipBytes = (Get-Item -LiteralPath $Path).Length; gzipSha256 = File-Hash $Path }
}
