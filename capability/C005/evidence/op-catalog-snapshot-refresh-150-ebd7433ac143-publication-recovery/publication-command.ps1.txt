function Run-Git([string[]]$Arguments) {
    $start = [Diagnostics.ProcessStartInfo]::new()
    $start.FileName = 'git'
    $start.WorkingDirectory = (Get-Location).Path
    $start.UseShellExecute = $false
    $start.RedirectStandardOutput = $true
    $start.RedirectStandardError = $true
    $start.ArgumentList.Add('--no-pager')
    foreach ($argument in $Arguments) { $start.ArgumentList.Add($argument) }
    $process = [Diagnostics.Process]::new()
    $process.StartInfo = $start
    try {
        if (-not $process.Start()) { throw 'Could not start git.' }
        $outTask = $process.StandardOutput.ReadToEndAsync()
        $errTask = $process.StandardError.ReadToEndAsync()
        $process.WaitForExit()
        $out = $outTask.GetAwaiter().GetResult()
        $err = $errTask.GetAwaiter().GetResult()
        if ($process.ExitCode -ne 0) { throw "git $($Arguments -join ' ') failed ($($process.ExitCode)): $out$err" }
        if ($err) { [Console]::Error.Write($err) }
        if ($out) { return ($out -split '\r?\n' | Where-Object { $_ -ne '' }) }
    } finally { $process.Dispose() }
}
function Owned-Path([string]$Path) {
    return $Path.Replace('/', '\') -match '^capability\\(?:C005|C006|C007|C008|local-harness)\\'
}
function Unowned-Index {
    return @(Run-Git @('ls-files', '--stage') | Where-Object { -not (Owned-Path (($_ -split "`t", 2)[1])) })
}
function Unowned-Flags {
    return @(Run-Git @('ls-files', '-v') | Where-Object { -not (Owned-Path ("$_".Substring(2))) })
}
function Text-Hash([string]$Text) {
    return [Convert]::ToHexString([Security.Cryptography.SHA256]::HashData(
        [Text.Encoding]::UTF8.GetBytes($Text))).ToLowerInvariant()
}
function Assert-Preserved {
    $historical = Get-Content -LiteralPath "$evidence\historical-proof-hashes.json" -Raw | ConvertFrom-Json
    foreach ($file in $historical.files) {
        if ((Get-FileHash -LiteralPath $file.path -Algorithm SHA256).Hash.ToLowerInvariant() -ne $file.sha256) {

    $prior = Get-Content -LiteralPath "$evidence\prior-attempt-frozen-hashes.json" -Raw | ConvertFrom-Json
    foreach ($directory in @($initial.blockedEvidence, $prior.path)) {
        $expected = @($historical.files | Where-Object { $_.path.StartsWith("$directory\") } | ForEach-Object path) | Sort-Object
        $actual = @(Get-ChildItem -LiteralPath $directory -Recurse -File -Force | ForEach-Object {
            [IO.Path]::GetRelativePath((Get-Location).Path, $_.FullName)
        }) | Sort-Object
        if (@(Compare-Object @($expected) @($actual)).Count) { throw "Frozen attempt file set changed: $directory" }
    }
    foreach ($file in @($initial.readOnlyBoundaries.publishedHelper, $initial.readOnlyBoundaries.standardPublisher,
        $initial.readOnlyBoundaries.adminStateBoundary)) {
        if ((Get-FileHash -LiteralPath $file.path -Algorithm SHA256).Hash.ToLowerInvariant() -ne $file.sha256) {
            throw "Read-only helper or standard/Admin boundary changed: $($file.path)"
        }
    }
    foreach ($file in $initial.exactRawJsonExceptions) {
        $logical = $file.path.Replace('\', '/')
        $raw = (Run-Git @('hash-object', '--no-filters', '--', $file.path) | Out-String).Trim()
        $filtered = (Run-Git @('hash-object', "--path=$logical", '--', $file.path) | Out-String).Trim()
        if ($raw -ne $file.gitObject -or $filtered -ne $raw) { throw "Canonical-path filtering changed raw JSON: $logical" }
        $attributes = @(Run-Git @('check-attr', 'text', 'diff', '--', $logical)) -join "`n"
        if ($attributes -notmatch ': text: unset' -or $attributes -notmatch ': diff: unset') {
            throw "An exact historical JSON exception is missing: $logical"
        }
    }
    $manifest = Get-Content -LiteralPath "$evidence\hashes-final.json" -Raw | ConvertFrom-Json
    foreach ($file in $manifest.files) {
        if ((Get-FileHash -LiteralPath $file.path -Algorithm SHA256).Hash.ToLowerInvariant() -ne $file.sha256) {
            throw "Verified current owned bytes changed: $($file.path)"
        }
    }
    $actualPaths = @(foreach ($scope in $scopes) {
        Get-ChildItem -LiteralPath $scope -Recurse -File -Force | ForEach-Object {
            [IO.Path]::GetRelativePath((Get-Location).Path, $_.FullName)
        }
    }) | Where-Object { $_ -ne "$evidence\hashes-final.json" } | Sort-Object
    if (@(Compare-Object @($manifest.files.path | Sort-Object) @($actualPaths)).Count) {
        throw 'Current complete owned inventory changed after verification.'
    }
}
function Verify-PrecursorTag {
    if ((Run-Git @('cat-file', '-t', $precursorTag) | Out-String).Trim() -ne 'tag' -or
        (Run-Git @('rev-parse', "$precursorTag^{}") | Out-String).Trim() -ne $precursor) {
        throw 'Existing unqualified marker must be annotated and identify the exact failed precursor.'
    }
    $body = (Run-Git @('for-each-ref', '--format=%(contents)', "refs/tags/$precursorTag") | Out-String)
    if ($body -notmatch '(?i)unqualified' -or $body -notmatch 'OP-CATALOG-SNAPSHOT-REFRESH-150' -or
        $body -notmatch '(?i)(zero|no)[^\r\n]*credit' -or
        $body -match '"accepted"\s*:\s*true|"parentAcceptanceClaimed"\s*:\s*true') {
        throw 'Existing failure marker meaning is not explicitly unqualified and zero-credit; never overwrite it.'
    }
}
