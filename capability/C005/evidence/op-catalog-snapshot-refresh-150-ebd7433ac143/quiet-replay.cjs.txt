function parseTap(text) {
  const summary = {};
  for (const key of ['tests', 'pass', 'fail', 'cancelled', 'skipped', 'todo']) {
    summary[key] = Number(new RegExp(`^# ${key} (\\d+)\\r?$`, 'm').exec(text)?.[1] || 0);
  }
  const stack = [];
  const parents = new Set();
  let total = 0;
  for (const line of text.split(/\r?\n/)) {
    const match = /^(\s*)# Subtest: /.exec(line);
    if (!match) continue;
    const indent = match[1].length;
    while (stack.length && stack.at(-1).indent >= indent) stack.pop();
    if (stack.length) parents.add(stack.at(-1).id);
    stack.push({ indent, id: total++ });
  }
  return { ...summary, total, parents: parents.size, leaves: total - parents.size };
}
