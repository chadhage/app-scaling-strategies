function Run-Git([string[]]$Arguments) {
    $start = [Diagnostics.ProcessStartInfo]::new()
    $start.FileName = 'git'
    $start.WorkingDirectory = (Get-Location).Path
    $start.UseShellExecute = $false
    $start.RedirectStandardOutput = $true
    $start.RedirectStandardError = $true
    $start.ArgumentList.Add('--no-pager')
    foreach ($argument in $Arguments) { $start.ArgumentList.Add($argument) }
    $process = [Diagnostics.Process]::new()
    $process.StartInfo = $start
    try {
        if (-not $process.Start()) { throw 'Could not start git.' }
        $stdoutTask = $process.StandardOutput.ReadToEndAsync()
        $stderrTask = $process.StandardError.ReadToEndAsync()
        $process.WaitForExit()
        $stdout = $stdoutTask.GetAwaiter().GetResult()
        $stderr = $stderrTask.GetAwaiter().GetResult()
        if ($process.ExitCode -ne 0) { throw "git $($Arguments -join ' ') failed ($($process.ExitCode)): $stdout$stderr" }
        if ($stderr) { [Console]::Error.Write($stderr) }
        if ($stdout) { return ($stdout -split '\r?\n' | Where-Object { $_ -ne '' }) }
    } finally {
        $process.Dispose()
    }
}
function Owned-Path([string]$Path) {
    return $Path.Replace('/', '\') -match '^capability\\(?:C005|C006|C007|C008|local-harness)\\'
}
function Unowned-Index {
    return @(Run-Git @('ls-files', '--stage') | Where-Object { -not (Owned-Path (($_ -split "`t", 2)[1])) })
}
function Unowned-IndexFlags {
    return @(Run-Git @('ls-files', '-v') | Where-Object { -not (Owned-Path ("$_".Substring(2))) })
}
function Text-Hash([string]$Text) {
    return [Convert]::ToHexString([Security.Cryptography.SHA256]::HashData(
        [Text.Encoding]::UTF8.GetBytes($Text))).ToLowerInvariant()
}
function Assert-Preserved {

    }

    $expectedIds = @(1..135 | ForEach-Object { 'S{0:D3}' -f $_ })

        $generation.acceptedPortableLessons -ne 135 -or $generation.sourceKanoDone.S -ne 148 -or
        $generation.sourceKanoDone.D -ne 2 -or $generation.sourceKanoDone.L -ne 0 -or
        @(Compare-Object @($expectedIds | Sort-Object) @($generation.acceptedPortableIds | Sort-Object)).Count) {
        throw 'Immutable source or exact accepted IDs/counts changed.'
    }
    if ($generation.generator.gitReadMaxBufferBytes -ne 8388608 -or
        $generation.generator.onlyFullShaPinChanged -ne $true -or
        $generation.generator.extractorSemanticsUnchanged -ne $true -or
        $generation.nativeRunnable -ne 0 -or $generation.nativeQualified -ne 0 -or
        $generation.completedSourceParents -ne 0 -or $generation.H001Complete -ne $false -or
        $generation.H002Complete -ne $false -or $generation.production24Validated -ne $false -or
        $generation.liveAvailability -ne $false -or
        $generation.S107StaticEntryCertifiesHttpOriginBehavior -ne $false -or
        $generation.recovery.timestampFalsification -ne $false -or
        $generation.recovery.validatorSchemaAuthorityOrChronologyRelaxation -ne $false) {
        throw 'Existing transport, exact chronology, local-only or authority boundaries changed.'
    }
    foreach ($snapshot in @($generation.original, $generation.intermediate, $generation.prior36,
        $generation.prior61, $generation.prior86, $generation.prior100, $generation.previous,
        $generation.data, $generation.recovery.failedGeneratedData)) {
        if ((Get-FileHash -LiteralPath $snapshot.path -Algorithm SHA256).Hash.ToLowerInvariant() -ne $snapshot.sha256) {
            throw "Current, historical or failed-source catalog bytes changed: $($snapshot.path)"
        }
    }
    $initial = Get-Content -LiteralPath "$evidence\initial-boundary.json" -Raw | ConvertFrom-Json
    foreach ($file in @($initial.readonlyExtra.publishedHelper, $initial.readonlyExtra.standardPublisher,
        $initial.readonlyExtra.adminStateBoundary)) {
        if ((Get-FileHash -LiteralPath $file.path -Algorithm SHA256).Hash.ToLowerInvariant() -ne $file.sha256) {
            throw "Read-only helper or standard/Admin publisher changed: $($file.path)"
        }
    }
    $manifest = Get-Content -LiteralPath "$evidence\hashes-final.json" -Raw | ConvertFrom-Json
    foreach ($file in $manifest.files) {
        if ((Get-FileHash -LiteralPath $file.path -Algorithm SHA256).Hash.ToLowerInvariant() -ne $file.sha256) {
            throw "Verified owned bytes changed before publication: $($file.path)"
        }
    }
    $currentPaths = @(foreach ($scope in $scopes) {
        Get-ChildItem -LiteralPath $scope -Recurse -File -Force | ForEach-Object {
            [IO.Path]::GetRelativePath((Get-Location).Path, $_.FullName)
        }
    }) | Where-Object { $_ -ne "$evidence\hashes-final.json" } | Sort-Object
    if (@(Compare-Object @($manifest.files.path | Sort-Object) @($currentPaths)).Count) {
        throw 'Owned source/proof inventory changed after the final manifest.'
    }
}
