function fingerprint(file, canonical = true) {
  const raw = fs.readFileSync(path.join(root, file));
  const result = { path: file, bytes: raw.length, sha256: sha(raw) };
  if (/\.(?:js|cjs|ts|txt|json|feature|css|html|md)$/.test(file) || file.endsWith('.gitattributes')) {
    result.normalizedSha256 = sha(normalize(raw));
    result.rawCRLFPairs = (raw.toString('utf8').match(/\r\n/g) || []).length;
  }
  if (canonical) {
    const committed = git(['show', `${sourceCommit}:${file.replaceAll('\\', '/')}`]);
    result.canonicalGitSourceSha256 = sha(committed);
    if (result.normalizedSha256) result.canonicalGitNormalizedSha256 = sha(normalize(committed));
  }
  return result;
}
