'use strict';

const test = require('node:test');
const assert = require('node:assert/strict');
const fs = require('node:fs');
const path = require('node:path');
const { createHash } = require('node:crypto');
const lesson = require('.\\model.js');
const root = __dirname;
const read = (name) => fs.readFileSync(path.join(root, name));
const json = (name) => JSON.parse(read(name).toString('utf8'));
const sha256 = (bytes) => createHash('sha256').update(bytes).digest('hex');

function ownedHashes(directory = root, prefix = '') {
  const hashes = {};
  for (const item of fs.readdirSync(directory, { withFileTypes: true }).sort((a, b) => a.name.localeCompare(b.name))) {
    const relative = path.join(prefix, item.name);
    if (item.isDirectory()) Object.assign(hashes, ownedHashes(path.join(directory, item.name), relative));
    else hashes[relative] = sha256(fs.readFileSync(path.join(directory, item.name)));
  }
  return hashes;
}
const before = ownedHashes();
test.after(() => assert.deepEqual(ownedHashes(), before, 'integrity verification must not change any owned bytes'));

test('raw source CRLF SHA256 and canonical LF Git identities are independently verifiable', () => {
  for (const source of json('evidence\\source-acceptance.json').sources) {
    const bytes = read(source.snapshot);
    assert.equal(bytes.length, source.bytes);
    assert.equal(sha256(bytes), source.sha256);
    assert.equal(source.hasCRLF, true);
    assert.ok(bytes.includes(Buffer.from('\r\n')));
    for (const excerpt of source.excerpts) {
      assert.ok(bytes.includes(Buffer.from(excerpt.rawLine, 'utf8')));
      assert.equal(sha256(Buffer.from(excerpt.rawLine, 'utf8')), excerpt.sha256);
    }
    const lf = Buffer.from(bytes.toString('utf8').replace(/\r\n/g, '\n'));
    assert.notEqual(sha256(lf), source.sha256);
    assert.equal(sha256(lf), source.canonicalLfSha256);
    const git = createHash('sha1').update(Buffer.from(`blob ${lf.length}\0`)).update(lf).digest('hex');
    assert.equal(git, source.canonicalGitBlobSha1);
  }
});

test('lesson has external local assets, restrictive CSP and no execution/storage/load-generator surface', () => {
  const html = read('index.html').toString('utf8');
  const code = read('app.js').toString('utf8') + read('model.js').toString('utf8');
  const csp = html.match(/http-equiv="Content-Security-Policy" content="([^"]+)"/)?.[1];
  assert.equal(csp, "default-src 'none'; script-src 'self'; style-src 'self'; connect-src 'none'; img-src 'none'; font-src 'none'; object-src 'none'; base-uri 'none'; form-action 'none'; frame-src 'none'; worker-src 'none'");
  assert.ok(html.indexOf('Content-Security-Policy') < html.indexOf('<script'));
  assert.deepEqual([...html.matchAll(/<script[^>]+src="([^"]+)"/g)].map((match) => match[1]), ['model.js', 'app.js']);
  assert.match(html, /href="styles.css"/);
  assert.ok(!/<(?:iframe|img|object|input)\b|(?:href|src)="(?:https?:|data:)|\bon\w+=|\bstyle=/.test(html));
  assert.ok(!/\b(?:fetch|XMLHttpRequest|WebSocket|Worker|setInterval|setTimeout|eval)\b|\b(?:localStorage|sessionStorage|indexedDB|cookie|innerHTML)\b/.test(code));
  assert.ok(!/\b(?:id|name)="reset"/.test(html));
  assert.ok(!/\bform\.reset\s*\(/.test(code));
  assert.match(html, /role="alert" tabindex="-1"/);
  assert.match(html, /B-S141 and D-S141 remain unverified/);
  assert.ok(html.includes(lesson.BANNER));
  assert.ok(read('steady-load.feature').toString('utf8').startsWith('Feature:'));
});

test('historical browser proof verifies captured bytes, not mutable current sources', () => {
  const directory = path.join(root, 'evidence', 'browser');
  if (!fs.existsSync(directory)) return;
  for (const name of fs.readdirSync(directory)) {
    assert.match(name, /^capture-[\w-]+$/);
    const proof = json(path.join('evidence', 'browser', name, 'observations.json'));
    assert.equal(proof.cardId, 'A-S141');
    assert.equal(proof.exactCaptureOptIn, true);
    assert.equal(proof.helperSha256, json('manifest.json').browserTooling.sha256);
    assert.ok(proof.sources.length >= 13);
    for (const source of proof.sources) {
      const bytes = read(path.join('evidence', 'browser', name, source.snapshot));
      assert.equal(sha256(bytes), source.sha256);
      assert.equal(bytes.length, source.bytes);
    }
    if (proof.outcome === 'passed') {
      assert.equal(proof.closedSessions, 2);
      assert.equal(proof.readOnlyBeforeCapture, true);
      assert.ok(proof.groups.length >= 9);
      assert.ok(proof.groups.every((group) => group.passed === true));
    } else {
      assert.equal(proof.outcome, 'failed');
      assert.ok(proof.failure?.message);
    }
  }
});

test('all retained child-run stdout/stderr bytes match their independent raw hashes', () => {
  const directory = path.join(root, 'evidence', 'raw');
  for (const name of fs.readdirSync(directory).filter((file) => file.endsWith('.summary.json'))) {
    const summary = json(path.join('evidence', 'raw', name));

    assert.equal(sha256(read(path.join('evidence', 'raw', `${prefix}.stderr.txt`))), summary.stderrSha256);
    assert.ok(summary.exitCode === null || Number.isInteger(summary.exitCode));
    if (summary.normalOwnedBytesUnchanged === true) {
      assert.equal(summary.guardFailure, null);
      assert.ok(summary.ownedFilesCompared > 0);
    }
  }
});
