const sha = (bytes) => createHash('sha256').update(bytes).digest('hex');

function tree(root) {
  const files = [];
  function visit(directory) {
    for (const entry of fs.readdirSync(directory, { withFileTypes: true })) {
      const file = path.join(directory, entry.name);
      if (entry.isDirectory()) visit(file);
      else {
        const stat = fs.statSync(file);
        files.push([path.relative(root, file), stat.size, stat.mtimeMs, sha(fs.readFileSync(file))]);
      }
    }
  }
  visit(root);
  return files.sort((left, right) => left[0].localeCompare(right[0]));
}

  } catch (error) {

      try {

      } catch (error) {
        record.domCaptureError = error.message;
      } finally {

      fs.writeFileSync(path.join(rawDirectory, 'dom-final.raw.html'), finalDOM);
      fs.writeFileSync(path.join(rawDirectory, 'console.raw.json'), JSON.stringify(record.console, null, 2) + '\n');
      console.log(`Deliberate raw browser capture: ${path.relative(process.cwd(), rawDirectory)}`);
    }
  }
  if (!capture) assert.deepEqual(tree(__dirname), before, 'Normal browser verification must not modify any lesson or historical proof file.');
  if (failure && cleanupFailure) throw new AggregateError([failure, cleanupFailure], 'Browser assertion and cleanup both failed.');
  if (failure) throw failure;
  if (cleanupFailure) throw cleanupFailure;
  assert.equal(record.cleanup.awaitedFinallyClose, true);
  assert.equal(record.cleanup.completed, true);

});
