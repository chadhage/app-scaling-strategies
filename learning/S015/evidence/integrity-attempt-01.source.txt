test('qualified RED raw output and non-discovered source snapshots are byte-preserved', () => {
  const proof = json('evidence\\red-provenance.json');
  assert.equal(proof.exitCode, 1);
  assert.equal(proof.assertionFailures, 46);
  assert.deepEqual(proof.summary, { tests: 47, pass: 1, fail: 46, skipped: 0 });
  assert.equal(hash(path.join(repo, proof.log.path)), proof.log.sha256);
  for (const snapshot of proof.snapshots) {
    assert.equal(hash(path.join(repo, snapshot.path)), snapshot.sha256);
    assert.ok(snapshot.path.endsWith('.source.txt'));
    assert.doesNotMatch(snapshot.path, /\.test\.(?:cjs|js|mjs)$/);

  const log = read('evidence\\red.tap.txt');
  assert.equal([...log.matchAll(/code: 'ERR_ASSERTION'/g)].length, 46);
  assert.doesNotMatch(log, /MODULE_NOT_FOUND|Cannot find module|ERR_TEST_FAILURE|name: 'TypeError'/);
  assert.equal(read('scale-in.test.cjs'), read('evidence\\red-tests.source.txt'));
  assert.equal(read('horizontal-scale-in.feature'), read('evidence\\red-bdd.source.txt'));
});

test('runtime is local-only, bounded, and protected by a restrictive non-inline CSP', () => {
  const html = read('index.html');
  const policy = html.match(/http-equiv="Content-Security-Policy" content="([^"]+)"/)[1];
  for (const directive of [
    "default-src 'none'", "script-src 'self'", "style-src 'self'", "connect-src 'none'",
    "img-src 'none'", "font-src 'none'", "media-src 'none'", "object-src 'none'",
    "base-uri 'none'", "form-action 'none'", "frame-src 'none'", "child-src 'none'",
    "worker-src 'none'", "manifest-src 'none'"
  ]) assert.ok(policy.includes(directive), directive);
  assert.doesNotMatch(policy, /unsafe-inline|unsafe-eval|https?:|\*/);
  assert.doesNotMatch(html, /\son[a-z]+\s*=|<style\b|\sstyle\s*=/i);
  const scripts = [...html.matchAll(/<script defer src="([^"]+)"><\/script>/g)].map((match) => match[1]);
  assert.deepEqual(scripts, ['scale-in.js', 'lesson.js']);
  for (const script of scripts) {
    assert.doesNotMatch(read(script), /\bfetch\s*\(|\bWebSocket\b|\bXMLHttpRequest\b|sendBeacon|localStorage|sessionStorage|document\.cookie|innerHTML|eval\s*\(|setTimeout|setInterval|\bDate\b|Math\.random/);
  }
  assert.doesNotMatch(read('lesson.css'), /@import|url\s*\(/i);
});

test('semantic controls are labelled, IDs are unique, errors announce, and form.reset is not shadowed', () => {
  const html = read('index.html');
  const ids = [...html.matchAll(/\bid="([^"]+)"/g)].map((match) => match[1]);
  assert.equal(new Set(ids).size, ids.length);
  const controls = [...html.matchAll(/<(?:input|select|button)\b[^>]*>/g)].map((match) => match[0]);
  assert.ok(controls.length >= 16);
  assert.ok(controls.every((control) => !/\b(?:name|id)="reset"/i.test(control)));
  for (const match of html.matchAll(/<(?:input|select)\b[^>]*\bid="([^"]+)"/g)) {
    assert.ok(html.includes(`for="${match[1]}"`), match[1]);
  }
  assert.match(html, /role="alert"/);
  assert.match(html, /role="status" aria-live="polite"/);
  assert.match(html, /<caption>Originally admitted synthetic work/);
  assert.match(html, /<noscript>/);
  assert.match(read('lesson.js'), /form\.reset\(\)/);
  assert.match(read('lesson.css'), /:focus-visible/);
  assert.equal(html.split('Training rehearsal - synthetic data - not a live Azure result').length - 1, 1);
});
